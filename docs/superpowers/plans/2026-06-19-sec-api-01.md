## Task 2: Refresh Deduplication in Frontend

**Problem:** If two API calls fail with 401 simultaneously, `handleApiError` in `useAuth.js` calls `refreshSession` twice. The second call uses an already-rotated token → 401 → logout. User loses session unexpectedly.

**Files:**
- Modify: `frontend-react/src/api.js` — add module-level dedup promise

- [ ] **Step 1: Add dedup module in api.js**

In `frontend-react/src/api.js`, after the existing imports and constants, add:

```js
// Deduplication: at most one in-flight refresh at a time
let _refreshPromise = null;

export async function refreshSession(refreshToken) {
  if (_refreshPromise) return _refreshPromise;
  _refreshPromise = api("/auth/refresh", {
    method: "POST",
    body: JSON.stringify({ refreshToken: refreshToken || "" }),
  }).finally(() => {
    _refreshPromise = null;
  });
  return _refreshPromise;
}
```

Find the existing `refreshSession` function in `api.js` (around line 114):
```js

  });
}
```

## Task 3: Standardize CPF/CNS Mask Pattern

**Current:** `maskSensitivePatientFields` sets `cpf = "***.***.***-**"` (no digits shown).
**Target:** Show last 2 digits of CPF, last 2 of CNS. Format: `***.***.**-XX` and `*** **** **** XXXX X` (simplified).

**Files:**
- Modify: `backend/src/utils/patients.js` — update `maskSensitivePatientFields`

- [ ] **Step 1: Write the masking functions**

In `backend/src/utils/patients.js`, add two helper functions before `maskSensitivePatientFields`:

```js
function maskCpf(raw) {
  const digits = String(raw || "").replace(/\D/g, "");
  if (digits.length === 11) return `***.***.***-${digits.slice(9)}`;
  if (raw) return "***.***.***-**";
  return raw;
}

function maskCns(raw) {
  const digits = String(raw || "").replace(/\D/g, "");
  if (digits.length === 15) return `***.***.***.*****-${digits.slice(13)}`;
  if (raw) return "***.***.***.*****-**";
  return raw;
}
```

- [ ] **Step 2: Update maskSensitivePatientFields**

Replace:
```js
function maskSensitivePatientFields(patient) {
  if (!patient) return patient;
  const masked = { ...patient };
  if (masked.cpf) masked.cpf = "***.***.***-**";
  if (masked.cns) masked.cns = "***.***.***.*****-**";
  return masked;
}
```
With:
```js
function maskSensitivePatientFields(patient) {
  if (!patient) return patient;
  const masked = { ...patient };
  if (masked.cpf) masked.cpf = maskCpf(masked.cpf);
  if (masked.cns) masked.cns = maskCns(masked.cns);
  return masked;
}
```

```js
// GET /patients/:id — single patient detail (team-scoped)
router.get("/patients/:id", sensitiveDataRateLimit, async (req, res) => {
  const id = String(req.params.id || "").trim();
  if (!id) return res.status(400).json({ error: "id obrigatório" });

  const role = canonicalRole(req.user?.role);
  if (role === "receptionist") {
    return res.status(403).json({ error: "Recepcionista não tem acesso ao detalhe do paciente" });
  }

  const db = await readDb();
  ensureDbShape(db);

  const patient = db.patients.find((p) => p.id === id);
  if (!patient) return res.status(404).json({ error: "Paciente não encontrado" });

  if (role !== "break_glass_admin" && patient.teamId !== req.user.teamId) {
    return res.status(403).json({ error: "Sem permissão para acessar este paciente" });
  }

  await withDb((auditDb) => {
    ensureDbShape(auditDb);
    addAuditLog(auditDb, req.user, "patient.detail_read", "patient", id, {
      outcome: "success",
      teamId: patient.teamId,
      isCrossTeam: patient.teamId !== req.user.teamId,
    });
  });

  return res.json(filterNis(filterGestorSpecialCategory(filterCnsResponsavel(maskSensitivePatientFields(patient), req.user), req.user), req.user));
});
```

## Task 6: Optimistic Locking via updatedAt

**Strategy:** On `PATCH` for critical entities, if the request body includes `updatedAt`, compare it against the stored value. If they differ, return 409. If `updatedAt` is absent from the request, proceed (backward-compat).

**Scope:** `PATCH /patients/:id/cadastro-individual` and `PATCH /households/:id`.

**Files:**
- Modify: `backend/src/routes/patients.js` (cadastro-individual handler)

```js
// Optimistic locking: if client sends updatedAt, verify no concurrent edit

}
```

```js
import { describe, it, before, after } from "node:test";
import assert from "node:assert/strict";
import { createServer } from "node:http";

const BASE = process.env.TEST_API_URL || "http://localhost:3001";

async function post(path, body, token = "") {
  const headers = { "Content-Type": "application/json" };
  if (token) headers.Authorization = `Bearer ${token}`;
  const res = await fetch(`${BASE}${path}`, {
    method: "POST",
    headers,
    body: JSON.stringify(body),
  });
  return { status: res.status, body: await res.json().catch(() => ({})) };
}

async function get(path, token = "") {
  const headers = {};
  if (token) headers.Authorization = `Bearer ${token}`;
  const res = await fetch(`${BASE}${path}`, { headers });
  return { status: res.status, body: await res.json().catch(() => ({})) };
}

let app, server, adminToken, patientId, householdId;

before(async () => {
  process.env.NODE_ENV = "test";
  process.env.JWT_SECRET = "test-secret-sec-api-01";
  process.env.ACCESS_TOKEN_EXPIRES_IN = "15m";
  const { default: appModule } = await import("../src/app.js");
  app = appModule;
  server = createServer(app);
  await new Promise((resolve) => server.listen(3001, resolve));

  // Login as admin to get token
  const login = await post("/auth/login", {

    password: process.env.SEED_ADMIN_PASSWORD || "Admin@12345",

after(async () => {
  await new Promise((resolve) => server.close(resolve));

describe("SEC-API-01: Token lifetime", () => {
  it("GET /me returns user with valid token", async () => {
    const { status } = await get("/me", adminToken);
    assert.equal(status, 200);
  });

  it("GET /me returns 401 with expired/invalid token", async () => {
    const { status } = await get("/me", "invalid.token.here");
    assert.equal(status, 401);
  });

  it("POST /auth/refresh with no token returns 400", async () => {
    const { status } = await post("/auth/refresh", {});
    assert.equal(status, 400);
  });
});

describe("SEC-API-01: GET /patients/:id", () => {
  before(async () => {
    // Use existing patient from list
    const { body } = await get("/patients", adminToken);
    patientId = Array.isArray(body) ? body[0]?.id : null;
  });

  it("returns 404 for unknown id", async () => {
    const { status } = await get("/patients/00000000-0000-0000-0000-000000000000", adminToken);
    assert.equal(status, 404);
  });

  it("returns 401 without token", async () => {
    if (!patientId) return;
    const { status } = await get(`/patients/${patientId}`);
    assert.equal(status, 401);
  });

  it("returns patient with masked CPF when id valid", async () => {
    if (!patientId) return;
    const { status, body } = await get(`/patients/${patientId}`, adminToken);
    assert.equal(status, 200);
    assert.ok(body.id, "patient has id");
    if (body.cpf) {
      assert.match(body.cpf, /^\*\*\*\.\*\*\*\.\*\*\*-/, "CPF should be masked");
    }
  });
});

describe("SEC-API-01: GET /households/:id", () => {
  before(async () => {
    const { body } = await get("/patients", adminToken);

    const { status } = await get("/households/00000000-0000-0000-0000-000000000000", adminToken);

  it("PATCH with stale updatedAt returns 409", async () => {

    const staleTs = "2020-01-01T00:00:00.000Z";
    const { status } = await post(`/households/${householdId}`, { updatedAt: staleTs });
    // This tests the backend rejects via method — PATCH is the actual verb
    const patchRes = await fetch(`${BASE}/households/${householdId}`, {
      method: "PATCH",
      headers: { "Content-Type": "application/json", Authorization: `Bearer ${adminToken}` },
      body: JSON.stringify({ updatedAt: staleTs, microarea: "01" }),

    assert.equal(patchRes.status, 409);
  });

      headers: { "Content-Type": "application/json", Authorization: `Bearer ${adminToken}` },
```
