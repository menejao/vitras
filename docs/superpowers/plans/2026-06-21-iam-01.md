# IAM-01: Platform Provisioning, National Console & First-Login Password Reset

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement national support console, UBS/team/user provisioning by support_admin, gestor local user management, forced first-login password change, and password reset — all with full audit trail and RBAC isolation between suporte nacional and dados clínicos.

**Architecture:** Add `support_admin` role (platform-only caps), `forcePasswordChange` field on all users, a new `/platform` Express router (guarded by `requireSupportAdmin`), a blocking middleware that prevents `support_admin` from reaching clinical routes, and two new frontend pages (`PlatformConsolePage` and `ChangePasswordRequiredPage`). Gestor local user creation is added to the existing `/users` route. All temp passwords are generated with `crypto.randomBytes`, hashed with scrypt before storage, and shown once on screen.

**Tech Stack:** Node.js + Express, JSON file DB (`withDb`/`readDb`), JWT (jwt.sign), React + Vite, `node:crypto`, `node:test` for tests.

function generateTempPassword() {
  // 14+ chars, mixed: upper + lower + digits + symbols, crypto-secure
  const upper  = "ABCDEFGHJKLMNPQRSTUVWXYZ";
  const lower  = "abcdefghjkmnpqrstuvwxyz";
  const digits = "23456789";
  const syms   = "!@#$%^&*";
  const all    = upper + lower + digits + syms;
  const bytes  = crypto.randomBytes(18);
  let password = [
    upper[bytes[0]  % upper.length],
    upper[bytes[1]  % upper.length],
    lower[bytes[2]  % lower.length],
    lower[bytes[3]  % lower.length],
    digits[bytes[4] % digits.length],
    digits[bytes[5] % digits.length],
    syms[bytes[6]   % syms.length],
    syms[bytes[7]   % syms.length],
  ];
  // fill up to 14 chars from full alphabet
  for (let i = 8; i < 14; i++) {
    password.push(all[bytes[i] % all.length]);
  }
  // Fisher-Yates shuffle using remaining bytes
  for (let i = password.length - 1; i > 0; i--) {
    const j = bytes[14 + (i % 4)] % (i + 1);
    [password[i], password[j]] = [password[j], password[i]];
  }
  return password.join("");
}
```

// IAM-01: Forced first-login password change
// Requires valid JWT (placed after requireAuth in app.js)
// Does NOT require CSRF — token is Bearer or cookie (CSRF already verified globally)
router.post("/auth/change-password-required", requireAuth, async (req, res) => {
  const { currentPassword, newPassword } = req.body || {};

  if (!currentPassword || !newPassword) {
    return res.status(400).json({ error: "currentPassword e newPassword são obrigatórios" });
  }

  if (!isStrongPassword(newPassword)) {
    return res.status(400).json({
      error: "Senha fraca: mínimo de 8 caracteres, 1 letra maiúscula, 1 número e 1 caractere especial"
    });
  }

  const result = await withDb((db) => {
    ensureDbShape(db);
    const idx = db.users.findIndex((u) => u.id === req.user.id);
    if (idx < 0) return { error: { status: 404, message: "Usuário não encontrado" } };

    const user = db.users[idx];

    // Verify current (temporary) password
    if (!verifyPassword(currentPassword, user.password)) {
      addAuditLog(db, req.user, "auth.change_password_required_failed", "auth", user.id, {
        outcome: "wrong_current_password"
      });
      return { error: { status: 401, message: "Senha atual incorreta" } };
    }

    // New password cannot be same as current temp password
    if (verifyPassword(newPassword, user.password)) {
      return { error: { status: 400, message: "Nova senha não pode ser igual à senha temporária" } };
    }

    // Check new password doesn't contain email, CPF-like sequences, or name
    const nameParts = String(user.name || "").toLowerCase().split(/\s+/).filter((p) => p.length > 3);
    const emailLocal = String(user.email || "").split("@")[0].toLowerCase();
    const newPwLower = newPassword.toLowerCase();
    if (nameParts.some((part) => newPwLower.includes(part))) {
      return { error: { status: 400, message: "Senha não pode conter partes do nome" } };
    }
    if (emailLocal && newPwLower.includes(emailLocal)) {
      return { error: { status: 400, message: "Senha não pode conter o e-mail" } };
    }

    const nowIso = new Date().toISOString();
    db.users[idx] = {
      ...user,
      password: hashPassword(newPassword),
      forcePasswordChange: false,
      passwordUpdatedAt: nowIso,
      temporaryPasswordIssuedAt: null,
      updatedAt: nowIso
    };

    // Invalidate all existing refresh tokens for this user
    db.refreshTokens = (db.refreshTokens || []).map((t) => {
      if (t.userId === user.id && !t.revokedAt) {
        return { ...t, revokedAt: nowIso };
      }
      return t;
    });

    addAuditLog(db, req.user, "auth.change_password_required_success", "auth", user.id, {
      outcome: "success"
    });

    return { user: db.users[idx], db };
  });

  if (result?.error) return res.status(result.error.status).json({ error: result.error.message });

  // Issue new session (old tokens were revoked)
  const latestDb = await readDb();
  const latestUser = latestDb.users.find((u) => u.id === req.user.id) || result.user;
  const refresh = await withDb((db2) => {
    ensureDbShape(db2);
    return createRefreshToken(latestUser.id, getClientIp(req), db2);
  });

  return res.json(issueSession(res, latestUser, latestDb, {}, refresh.raw, refresh.sessionId));
});
```

function requireSupportAdmin(req, res, next) {
  if (canonicalRole(req.user?.role) !== "support_admin") {
    return res.status(403).json({ error: "Acesso restrito ao Console Nacional" });
  }
  return next();
}

// Prevents support_admin from reaching any clinical route
// (defense-in-depth — capabilities already deny access, but this gives an explicit 403)
function blockSupportAdminFromClinical(req, res, next) {
  if (canonicalRole(req.user?.role) !== "support_admin") return next();
  // support_admin may only access /platform and /auth/* routes
  const allowedPrefixes = ["/platform", "/auth/", "/health", "/readyz", "/api-docs"];
  const allowed = allowedPrefixes.some((prefix) => req.path.startsWith(prefix));
  if (!allowed) {
    return res.status(403).json({ error: "Conta de suporte nacional não tem acesso a dados clínicos ou operacionais da UBS" });
  }
  return next();
}
